+        // The URL health check is skipped in the rest of the suite via
+        // WORKTRUNK_TEST_SKIP_URL_HEALTH_CHECK, so nothing else exercises
+        // `probe_addr`. Bind real loopback listeners to run both socket
+        // domains (IPv4 and IPv6) without depending on random local processes.
+        let v4 = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
+        assert!(probe_addr(v4.local_addr().unwrap()));
+
+        let v6 = std::net::TcpListener::bind("[::1]:0").unwrap();
+        assert!(probe_addr(v6.local_addr().unwrap()));
+    }
